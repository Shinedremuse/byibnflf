        [[ "$fext" == "$ext" ]] && match=1 && break
    done
    if [[ $match -eq 0 ]]; then
        for kw in "${SENSITIVE_KW[@]}"; do
            [[ "$flower" == *"$kw"* ]] && match=1 && break
        done
    fi
